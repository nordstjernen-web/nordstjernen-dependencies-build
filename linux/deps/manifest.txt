# Nordstjernen Linux prebuilt-dependency manifest
#
# Pinned third-party native dependencies that are compiled on Ubuntu into the
# Linux "GTK overlay" consumed when testing the desktop Nordstjernen engine.
#
# One record per line, whitespace separated:
#
#   <name>  <version>  <sha256>  <url>
#
# - <name>    must match the build function name in build-linux-deps.sh
# - <sha256>  is verified against the downloaded tarball before extraction.
#             The literal value AUTO means "not yet pinned": CI downloads the
#             tarball, prints its real sha256 (a `PINME <name> <sha>` line) and
#             proceeds WITHOUT verifying. AUTO is a LOCAL convenience only: the
#             build-linux-deps workflow rejects it on every branch. Before
#             pinning, verify the checksum against upstream's own .sha256sum /
#             signature (or a second packager), never just the PINME line.
# - lines beginning with '#' and blank lines are ignored
#
# Unlike the android/ and ios/ sysroots (which cross-compile the whole
# GLib/cairo/pango stack because mobile drops system libraries), the desktop
# Linux port builds *against* Ubuntu's system graphics stack. This manifest
# pins only what the development GTK needs and the distro does not ship new
# enough: GTK itself, the GLib and Pango releases it requires, and a newer
# wayland-protocols. Everything else (cairo, gdk-pixbuf, graphene, libepoxy,
# harfbuzz, fontconfig, wayland, ...) comes from the distro via
# `apt build-dep gtk4`.
#
# Versions below were pinned against upstream release tarballs and their
# checksums verified at pin time. Bump deliberately and re-verify the sha256.

# --- GTK build deps too new for the distro ----------------------------------
# GTK 4.23.x requires wayland-protocols >= 1.48, but even ubuntu-26.04 only
# ships 1.47. wayland-protocols is a tiny, XML-only meson package (no compiled
# code), so we build a new-enough release into our prefix; it installs
# wayland-protocols.pc into share/pkgconfig, which our prefix puts ahead of the
# system copy so GTK picks it up.
# Pinned to 1.48, not the newer 1.49: 1.49's staging/color-management XML fails
# `wayland-scanner --strict` DTD validation with the wayland 1.24 that
# ubuntu-26.04 ships (it needs a newer wayland-scanner), and GTK only needs
# >= 1.48. Re-bump when the build host moves to a distro with wayland >= 1.26.
wayland-protocols 1.48 c563af8e2e784f9599fe23819a3fc5e7d946e76db9f90a2e3ba6c9a869a52911  https://gitlab.freedesktop.org/wayland/wayland-protocols/-/archive/1.48/wayland-protocols-1.48.tar.gz

# --- GLib / Pango too new for the distro -----------------------------------
# GTK 4.25 requires glib >= 2.89.3 and pango >= 1.58; ubuntu-26.04 ships glib
# 2.88 and pango 1.57. Both are built into the overlay (same releases the mobile
# sysroots pin; GLib is ABI-backward-compatible, so the system cairo/gdk-pixbuf/
# harfbuzz keep working against it once the overlay is first on LD_LIBRARY_PATH).
# sha256 verified against the .sha256sum files GNOME publishes next to each tarball.
glib        2.90.1    93c941aa17d5eb1d53fe838365f29a8b4e539c222a256d974ec8f30fc413e396  https://download.gnome.org/sources/glib/2.90/glib-2.90.1.tar.xz
pango       1.58.2    342385b6ca3b7c73455d7c80a13b7dbe4489e00bc3bd4c5bd6ed4dce421e374a  https://download.gnome.org/sources/pango/1.58/pango-1.58.2.tar.xz

# sha256 verified against GNOME's gtk-4.25.0.sha256sum.
